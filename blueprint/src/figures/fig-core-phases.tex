% Round-loop per-process phase machine (PLTS.ABA.Composition.RoundLoopStep). Automata style.
% The edges drawn are the transitions that write the phase; the boxes below list the
% transitions that leave the phase where it stands.
\begin{tikzpicture}[
    ->, >={Stealth[round]}, auto, semithick, node distance=2.9cm, on grid,
    every state/.style={draw, ellipse, minimum width=1.8cm, minimum height=0.9cm,
      inner sep=1pt, align=center, font=\small},
    lbl/.style={font=\footnotesize, fill=white, inner sep=1.5pt},
    elbl/.style={font=\scriptsize, fill=white, inner sep=1.2pt}]
  \node[state, initial, initial text={}] (idle) {idle};
  \node[state] (tcg) [right=of idle] {\texttt{toCallG}};
  \node[state] (awg) [right=of tcg]  {\texttt{awaitG}};
  \node[state] (tcw) [right=of awg]  {\texttt{toCallW}};
  \node[state] (aww) [right=of tcw]  {\texttt{awaitW}};
  \path
    (idle) edge node[elbl, above]{\texttt{input}} (tcg)
    (tcg)  edge node[elbl, above]{\texttt{callG}} (awg)
    (awg)  edge node[elbl, above]{\texttt{retG}} (tcw)
    (tcw)  edge node[elbl, above]{\texttt{callW}} (aww)
    % inputLoop sits at every phase but idle; it is drawn at toCallG, on the
    % upper left so that the round back-edge can come down on the upper right
    (tcg)  edge[loop, out=110, in=150, looseness=7]
      node[lbl, above left, text width=1.6cm, align=flush center]{\texttt{inputLoop}} (tcg);
  % the DECIDED send of a grade-2 round, above the row between awaitG and toCallW
  \node[state] (tsd) at (7.25,2.1) {\texttt{toSendDecided}};
  \draw[->] (aww) to[out=110, in=0]
    node[elbl, left, pos=0.25, xshift=-3pt, text width=1.9cm, align=flush right]{\texttt{retW}: round${+}1$,\\ grade $2$} (tsd);
  \draw[->] (tsd) to[out=180, in=70]
    node[elbl, right, pos=0.6, xshift=3pt]{\texttt{decidedSend}} (tcg);
  % round back-edge, routed over the top so it clears the toSendDecided node
  \draw[->] (aww) to[out=80, in=100, looseness=1.0]
    node[lbl, above]{\texttt{retW}: round${+}1$} (tcg);
  % the transitions that leave the phase where it stands, below the row
  \node[draw, dashed, rounded corners, align=flush left, font=\scriptsize,
        text width=6.2cm, inner sep=4pt, anchor=north west] (dec) at (0.9,-1.15)
    {\textbf{the DECIDED transitions}, at every phase (D12$'$):\\
     \texttt{decidedRelay} under $f{+}1$ and $b \notin decidedSent$;\\
     \texttt{decidedDeliverReceive} at $b \notin decidedDelivered(k)$,
     \texttt{decidedDeliverLoop} at $b \in decidedDelivered(k)$;\\
     \texttt{ret} on $\mathrm{retABA}$ under $n{-}f$ and $b \in decidedSent$};
  \node[draw, dashed, rounded corners, align=flush left, font=\scriptsize,
        text width=4.0cm, inner sep=4pt, anchor=north west] (loops) at (7.75,-1.15)
    {\textbf{the loops}, at every phase:\\
     \texttt{retGLoop}, \texttt{retWLoop} take a return at a phase or a round that does not wait for it;\\
     \texttt{boundGIdle} takes $\mathrm{boundG}(r,b)$};
\end{tikzpicture}
