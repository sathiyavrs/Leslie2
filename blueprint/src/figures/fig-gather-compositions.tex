% The gather-based round and its three substitutions
% (PLTS.ABA.GBCA.ByAFW.roundOverBracha, roundOverBroadcastSpecification,
% roundOverGatherSpecifications, PLTS.ABA.GBCA.specificationOverRoundAlphabet).
% The bottom composition is drawn open: n programs beside the round's network, in
% parallel with two gather instances, each n programs beside the gather network
% in parallel with n Bracha instances, so 2n + 2 networks in one round. The round's
% network records the first gather's core from the gather's announcement and announces
% the round's bound bit (GBCA.ByAFW.NetworkStep); the gather network writes the core at
% the instance's first return and announces it (Gather.NetworkStep.announceCore).
\begin{tikzpicture}[
    >={Stealth[round]}, semithick,
    composition/.style={draw, rounded corners, align=center, font=\scriptsize,
      minimum width=10.4cm, minimum height=0.6cm, inner sep=3pt},
    prog/.style={draw, rounded corners, align=flush center, font=\scriptsize,
      minimum height=0.55cm, inner sep=2pt},
    brb/.style={draw, align=center, font=\scriptsize,
      minimum width=2.1cm, minimum height=0.5cm, inner sep=2pt},
    brbgap/.style={draw, align=center, font=\scriptsize,
      minimum width=0.55cm, minimum height=0.5cm, inner sep=1pt},
    lbl/.style={font=\scriptsize, align=left, fill=white, inner sep=1.5pt}]
  % ---- the bottom composition, drawn open
  \draw[dashed, rounded corners] (-0.35,-6.35) rectangle (12.25,-0.15);
  \node[font=\scriptsize, anchor=west] at (-0.2,-0.45)
    {$\mathrm{roundOverBracha}(P,r)$: programs, the round's network, two gather instances, all concrete};
  % the round's programs and network
  \node[prog, anchor=north west, text width=5.0cm] at (0.05,-0.75)
    {graded-agreement programs $1 \cdots n$:\\ input, candidate, outcome, return flag;\\
     outputs $\mathrm{secondGatherCall}$ and $\mathrm{retG}\,r$};
  \node[prog, anchor=north west, text width=5.6cm] at (6.25,-0.75)
    {the round's network, no messages:\\ the first core, from $\mathrm{firstGatherCore}\,C$;\\
     whether the round has returned;\\ outputs $\mathrm{boundG}\,r$ once both are held (D29)};
  % the two gathers
  \foreach \x/\t in {0.05/{gather 1, payloads \textsf{Bool}}, 6.25/{gather 2, payloads \textsf{Option Bool}}} {
    \draw[dashed, rounded corners] (\x,-5.05) rectangle ({\x+5.8},-2.35);
    \node[font=\scriptsize, anchor=north west] at ({\x+0.05},-2.4) {\t};
    \node[prog, anchor=north west, text width=2.6cm] at ({\x+0.1},-2.75)
      {gather programs\\ $1 \cdots n$, with\\ their variables};
    \node[prog, anchor=north west, text width=2.6cm] at ({\x+3.0},-2.75)
      {the gather network:\\ $sent$, $F$, the core;\\ outputs $\mathrm{deliver}$,\\ $\mathrm{core}\,C$ (D29)};
    \node[brbgap, anchor=north] at ({\x+2.9},-4.15) {$\cdots$};
    \node[brb, anchor=north east] at ({\x+2.5},-4.15) {$\mathrm{inputBroadcasts}\,1$};
    \node[brb, anchor=north west] at ({\x+3.3},-4.15) {$\mathrm{inputBroadcasts}\,n$};
    \node[font=\scriptsize, anchor=north west] at ({\x+0.05},-4.62)
      {$n$ Bracha instances};
  }
  \node[font=\scriptsize, align=flush center, text width=11.6cm] at (5.95,-5.7)
    {each Bracha instance $n$ programs beside its network; $2n + 2$ networks in one round\\
     the round's events, hidden: $\mathrm{firstGatherReturn}$, $\mathrm{firstGatherCore}$,
     $\mathrm{secondGatherCall}$, $\mathrm{secondGatherReturn}$};
  % ---- the three compositions above it
  \node[composition] (overbrb) at (5.95,-7.1)
    {$\mathrm{roundOverBroadcastSpecification}(P,r)$: each Bracha instance replaced by the BRB specification};
  \node[composition] (overgather)  at (5.95,-8.3)
    {$\mathrm{roundOverGatherSpecifications}(P,r)$: each gather instance replaced by the gather specification};
  \node[composition] (specification)  at (5.95,-9.5)
    {$\mathrm{GBCA.specificationOverRoundAlphabet}(P,r)$: Transition System 2 over the round's alphabet};
  \draw[->] (5.95,-6.35) -- (overbrb.north)
    node[lbl, midway, anchor=west, xshift=4pt]{$\mathrm{broadcastSubstitution}$, by congruence};
  \draw[->] (overbrb) -- (overgather)
    node[lbl, midway, anchor=west, xshift=4pt]{$\mathrm{gatherSubstitution}$, by congruence};
  \draw[->] (overgather) -- (specification)
    node[lbl, midway, anchor=west, xshift=4pt]{$\mathrm{refinesSpecification}$, along $\mathrm{SpecificationRelation}$};
\end{tikzpicture}
