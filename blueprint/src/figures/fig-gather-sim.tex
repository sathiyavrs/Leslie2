% The three tier simulations of one gather-based round
% (PLTS.ABA.GBCA.ByAFW.broadcastSubstitution, gatherSubstitution,
% refinesSpecification).
% Concrete rows above each tier, the tier below, dashed verticals the relation.
\begin{tikzpicture}[
    >={Stealth[round]}, semithick, on grid,
    conc/.style={draw, rounded corners, minimum height=0.8cm, align=center,
      font=\footnotesize, inner sep=3pt},
    lvl/.style={draw, rounded corners, minimum width=3.1cm, minimum height=0.8cm,
      align=center, font=\footnotesize, inner sep=3pt},
    note/.style={font=\scriptsize, align=left},
    rel/.style={font=\scriptsize, align=center}]
  \node[lvl] (overbracha) at (0,1.95)  {$\mathrm{roundOverBracha}$};
  \node[lvl] (overbrb)    at (0,0.35)  {$\mathrm{roundOverBroadcastSpecification}$};
  \node[lvl] (overgather) at (0,-1.25) {$\mathrm{roundOverGatherSpecifications}$};
  \node[lvl] (spec)  at (0,-2.85) {$\mathrm{GBCA.specInst}$};
  \draw[->, thick] (overbracha) -- (overbrb);
  \draw[->, thick] (overbrb)    -- (overgather);
  \draw[->, thick] (overgather) -- (spec);
  \node[conc] (r1) at (9.4,1.15)
    {every $\mathrm{ECHO}$, $\mathrm{VOTE}$, delivery\\ and return of a Bracha instance};
  \node[conc] (r2) at (9.4,-0.45)
    {every $\mathrm{ECHO}$, $\mathrm{VOTE}$ and $\mathrm{BIND}$\\ of a gather instance};
  \node[conc] (r3) at (9.4,-2.05)
    {a return whose exclusion is missing;\\ a return whose exclusion has happened};
  \draw[dashed, ->] (r1) -- (2.45,1.15) node[rel, pos=0.5, above]{$\mathrm{BroadcastSubstitutionRelation}$};
  \draw[dashed, ->] (r2) -- (2.45,-0.45) node[rel, pos=0.5, above]{$\mathrm{GatherSubstitutionRelation}$};
  \draw[dashed, ->] (r3) -- (2.45,-2.05) node[rel, pos=0.5, above]{$\mathrm{SpecificationRelation}$};
  \node[note, anchor=west] at (12.3,1.15)  {a stutter or a return\\ of the BRB specification};
  \node[note, anchor=west] at (12.3,-0.45) {a stutter; a broadcast\\ return writes a record};
  \node[note, anchor=west] at (12.3,-2.05) {a run, or a single\\ graded return};
  \node[font=\footnotesize, align=center] at (7.2,-3.85)
    {each stage idealises away one tier and holds the tiers above it equal;\\
     the two upper stages are congruence applications, the round's network held equal,\\
     and $\mathrm{SpecificationRelation}$ pins $excluded$ to the complement of the bit it holds};
\end{tikzpicture}
