% GBCA specification instance (PLTS.ABA.GBCA.Step). Automata style.
% Nodes are the reachable states quotiented to the pair (|excluded|, grade). The
% bit-labelled reachable states are (empty, bot), ({b}, bot), ({b}, A) and
% ({b}, C) for each bit b, seven in all. (empty, A) and (empty, C) are
% unreachable, every graded return needing excluded nonempty; and excluded never holds
% two bits, bindUnset being enabled from excluded = empty alone, so no second exclusion
% is drawn. The exclude edge names the bit it excludes, so 1-b is the surviving bit
% at every node to its right. Every return announces the round's bound bit bnd,
% the complement of the excluded one, so bnd = 1-b at every node to the right of
% the exclude edge. call/callLoop are enabled in every state and
% drawn once, on the initial state; fail is omitted.
\begin{tikzpicture}[
    ->, >={Stealth[round]}, auto, semithick, on grid,
    every state/.style={draw, minimum size=1.45cm, align=center, font=\small},
    lbl/.style={font=\footnotesize, align=center, fill=white, inner sep=1.5pt}]

  \node[state]                           (dA) at (4.9,3.0)  {$(1,2)$};
  \node[state, initial, initial text={}] (e0) at (0,0)      {$(0,\bot)$};
  \node[state]                           (d0) at (4.9,0)    {$(1,\bot)$};
  \node[state]                           (dC) at (4.9,-3.0) {$(1,0)$};

  % self-loops
  \path
    (e0) edge[loop below, looseness=5]
      node[lbl, below]{1 \texttt{call}\\2 \texttt{callLoop}} (e0)
    (d0) edge[loop, out=-28, in=28, looseness=5]
      node[lbl, right]{4 \texttt{retGrade1}\ $1{-}b$\\$bnd = 1{-}b$} (d0)
    (dA) edge[loop above, looseness=5]
      node[lbl, above]{4 \texttt{retGrade1}\ $1{-}b$\\5 \texttt{retGrade2}\ $1{-}b$\\$bnd = 1{-}b$} (dA)
    (dC) edge[loop below, looseness=5]
      node[lbl, below]{4 \texttt{retGrade1}\ $1{-}b$\\6 \texttt{retGrade0}\\$bnd = 1{-}b$} (dC);

  % the exclusion
  \path
    (e0) edge node[lbl, above]{3 \texttt{bindUnset}\ $b$} (d0);

  % the two grade guards
  \path
    (d0) edge node[lbl, left]{5 \texttt{retGrade2}\ $1{-}b$\\$bnd = 1{-}b$} (dA)
    (d0) edge node[lbl, left]{6 \texttt{retGrade0}\\$bnd = 1{-}b$} (dC);

  % the note on the announced bit
  \node[font=\footnotesize, align=center, text width=10.5cm] at (2.45,-5.75)
    {every return announces the bound bit $bnd$, the complement of the excluded bit (D29)};
\end{tikzpicture}
