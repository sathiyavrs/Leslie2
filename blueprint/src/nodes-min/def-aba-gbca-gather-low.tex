\begin{definition}[The gather-based GBCA implementation]
  \label{def:aba-gbca-gather-low}
  \lean{PLTS.ABA.GBCA.IdealState}\lean{PLTS.ABA.GBCA.idealInst}\lean{PLTS.ABA.GBCA.LowPairState}\lean{PLTS.ABA.GBCA.lowPairInst}\leanok
  \uses{def:aba-round-sub, def:aba-gbca-pair, def:aba-gather-ideal, def:aba-gather-low}
  The round of Definition~\ref{def:aba-round-sub} at its two concrete tiers, by the gather system in its two gather slots: over gather instances of Definition~\ref{def:aba-gather-ideal}, and fully concrete over gather instances of Definition~\ref{def:aba-gather-low} --- the two gather instances and, beneath them, the $4n$ Bracha instances carrying the inputs and the $\mathrm{BIND}$ payloads.

\end{definition}
