\begin{theorem}[The gather-based composition]
  \label{thm:aba-afw-protocol-composed}
  \lean{PLTS.ABA.AFW.roundProjection}\lean{PLTS.ABA.AFW.BoundInvariant}\lean{PLTS.ABA.AFW.ProtocolRelation}\lean{PLTS.ABA.AFW.protocolSimulation}\lean{PLTS.ABA.AFW.protocol_composed}\leanok
  \uses{def:aba-afw-protocol, def:aba-afw-round-families, thm:forward-sound, def:achievable}
  The gather-based protocol is forward-simulated by its composed system, along the Dirac lift of a relation that computes every round of the composed family from the implementation's state, \leandecl{PLTS.ABA.AFW.roundProjection}, takes the rounds in the family's map to be the started rounds (D37), and carries what the projection needs of the ghost, \leandecl{PLTS.ABA.AFW.BoundInvariant}; hence $\operatorname{ATD}(\afw{\mathrm{protocol}}(P)) \subseteq \operatorname{ATD}(\afw{\mathrm{composed}}(P))$: \leandecl{PLTS.ABA.AFW.protocolSimulation}, \leandecl{PLTS.ABA.AFW.protocol_composed}.
  Each transition of the implementation is matched by a transition of the composed system on the same label, by a silent run of one round, or by the empty run.
\end{theorem}

\begin{proof}\leanok
  \uses{def:aba-afw-protocol, def:aba-afw-round-families, def:forwardsim}
  See the Lean proof of \leandecl{PLTS.ABA.AFW.protocolSimulation}.
\end{proof}
