\begin{definition}[The protocol]
  \label{def:aba-protocol}
  \lean{PLTS.System.mapIdle}\lean{PLTS.ABA.Implementation.coinLabelMap}\lean{PLTS.ABA.Composition.coinOverRoundAlphabet}\lean{PLTS.ABA.ABDY.protocolExtended}\lean{PLTS.ABA.ABDY.protocolHidden}\lean{PLTS.ABA.ABDY.protocol}\leanok
  \uses{def:aba-network, def:aba-programs, def:aba-implementation, def:aba-wcc-spec, def:aba-labels, def:relabel, def:synchronised-product, def:process-algebra}
  The common coin $\mathrm{WCC.specFamily}(P)$ has the alphabet $\mathrm{Label}(n)$, so it is read over the extended alphabet by the construction $\mathrm{mapIdle}$ along the partial map $\mathrm{coinLabelMap}$, which sends the Byzantine coin call and return to the coin's own call and return and every other network event, the DECIDED send among them, to none: a label with an image steps by that image's transitions, and a label without one is a self-loop.
  The protocol is built in three stages,
  \[
    \abdy{\mathrm{protocolExtended}}(P) \;=\;
      \bigl(\mathrm{synchronisedProduct}_{j \in [n]}\ \mathrm{ABAProgram}(P,j)\bigr)
        \parallel \bigl(\mathrm{network}(P) \parallel \mathrm{coinOverRoundAlphabet}(P)\bigr),
  \]
  \[
    \abdy{\mathrm{protocolHidden}}(P) \;=\;
      \mathrm{relabel}\bigl(
        \mathrm{abstract}_{\mathrm{networkEventLabels}}(\abdy{\mathrm{protocolExtended}}(P))\bigr),
  \]
  \[
    \abdy{\mathrm{protocol}}(P) \;=\; \mathrm{abstract}_H\bigl(\abdy{\mathrm{protocolHidden}}(P)\bigr),
  \]
  where $\mathrm{coinOverRoundAlphabet}(P)$ is the common coin read along $\mathrm{coinLabelMap}$ and $H$ is the sub-protocol API of Definition~\ref{def:aba-labels}.
  This is the pipeline of Definition~\ref{def:aba-implementation} at ABDY22's implementation.
  The call records the input $b$ and sends nothing; the multicast of $\langle \mathrm{INPUT}, b \rangle$ is a round transition of its own.
  Hiding $\mathrm{networkEventLabels}$ discharges the extended alphabet, so the alphabet of $\abdy{\mathrm{protocolHidden}}(P)$ is exactly $\mathrm{Label}(n)$.
  Full construction: \leandecl{PLTS.ABA.ABDY.protocol}.
\end{definition}
